\section{Two errors}      % ARC 2 (~4 min)

\begin{frame}{Two ways to be wrong}
\begin{itemize}\makebiggaps
\item<1-> \textbf{Commission} --- assert an effect that isn't defensible.
\item<2-> \textbf{Omission} --- fail to assert an effect that \emph{is} defensible.
\end{itemize}
\vfill
% NB: evoke Type I / II, do NOT label them — room will hear the ordinary ones.
\onslide<3->{\textbf{We police the first. We are structurally blind to the second.}}
\end{frame}

\begin{frame}{Both live in \emph{your} work, not just medicine}
\begin{itemize}\makebiggaps
\item<1-> Commission: the observational estimate you report with a straight face.
\item<2-> Omission: the effect you \emph{could} defend, dropped the moment a
      referee says ``but it's not an experiment.''
\end{itemize}
\end{frame}

\section{The engine}      % ARC 5 (~13 min)

\begin{frame}{One generative question}
\centering\large
Where can we defensibly \textbf{know}, \textbf{bound}, or \textbf{model}
the counterfactual $\YY$ --- without an experiment?
\end{frame}

% ---- METHODS SPINE: they just did 2 days of POs/DAGs. Ground it in notation. ----
\begin{frame}{The estimand, in your notation}
\small
\[ \ATT \;=\; \E\!\left[\,Y_i(1) - Y_i(0)\;\middle|\; D_i = 1\,\right] \]
\begin{itemize}\makebiggaps
\item For a treated unit: $Y_i(1)$ is \emph{observed}; $Y_i(0)$ is the \emph{missing}
      counterfactual.
\item Estimator: $\displaystyle \att = \frac{1}{n_1}\sum_{i:\,D_i=1}
      \big[\,Y_i - \widehat{Y_i(0)}\,\big].$
\end{itemize}
\vfill
\onslide<2->{\textbf{Every design is just a claim about $\widehat{Y_i(0)}$.}
(Thesis 4.) Everything else is bookkeeping.}
\end{frame}

\begin{frame}{Safe inference in one line}
\centering\large
Declare a defensible interval for $\delta$ (the residual bias),\\[.3em]
then report the \emph{entire} ATT band across that interval.
\vfill
\onslide<2->{\normalsize Sensitivity analysis, partial identification, and SCQE
are all instances of this move.}
\end{frame}

% ---- The same move, worked mechanically at three levels of knowledge of Y(0) ----
\begin{frame}{Case 1 --- $\YY$ (nearly) known}
\small
\begin{itemize}\makebiggaps
\item Untreated course near-deterministic (e.g.\ uniformly fatal):
      $Y_i(0)\approx y_0$, essentially known.
\item Then $\displaystyle \att = \bar Y_{\text{treated}} - y_0$ --- a
      \emph{single treated arm} suffices.
\item No control group needed: the counterfactual was never in doubt.
\end{itemize}
\vfill
\onslide<2->{This is the GBM / melanoma regime. Randomization would buy nothing.}
\end{frame}

\begin{frame}{Case 2 --- $\YY$ uncertain: carry a $\delta$}
\small
Write the imputation \emph{with its own error}, and propagate it:
\[ \widehat{Y_i(0)} = m_i + \delta
   \qquad\Longrightarrow\qquad
   \ATT(\delta) = \bar Y_{\text{treated}} - \bar m - \delta . \]
\begin{itemize}\makebiggaps
\item $m_i$: best baseline guess. $\delta$: the residual bias you cannot rule out.
\item Defensible $\delta \in [\delta_{\text{lo}},\,\delta_{\text{hi}}]
      \;\Rightarrow\;$ an ATT \emph{band}, not a point.
\item Whole band clears $0$? \textbf{Defensible claim} (condition 2).
      Straddles $0$? Then say so --- that's the honest answer.
\end{itemize}
\end{frame}

\begin{frame}{The band, drawn --- condition 2 is a \emph{picture}}
\centering
\begin{tikzpicture}[scale=0.92]
  % axes
  \draw[->] (-0.3,0) -- (7.6,0) node[below right] {$\delta$};
  \draw[->] (0,-0.6) -- (0,4.9) node[above] {$\ATT(\delta)$};
  \node[below left] at (0,0) {$0$};
  % ATT(delta) = 4 - 0.5 delta  (linear; slope -1 in Case 2, -1/Delta-pi in SCQE)
  \draw[very thick, someblue] (0,4) -- (7,0.5);
  \node[someblue, right] at (5.1,1.9) {$\ATT(\delta)$};
  % defensible interval on delta-axis
  \draw[dashed, gray] (1,0) -- (1,3.5) -- (0,3.5);
  \draw[dashed, gray] (5,0) -- (5,1.5) -- (0,1.5);

  % resulting ATT band on the vertical axis

\vfill
{\footnotesize Band sits entirely above $0$ $\Rightarrow$ the sign is safe to assert.
Widen the $\delta$ interval until it touches $0$: that is exactly how much
ignorance you can tolerate before you must say ``we can't tell.''}
% Slope is $-1$ in Case 2; $-1/\Delta\pi$ in SCQE. Same figure, relabel the axis.
\end{frame}

% Notation now matches Wulf, Hazlett et al. 2025 (Observational Studies) / ps200c 11-scqe.
\begin{frame}{SCQE --- the intuition (two cohorts)}
\small
\begin{columns}[T]
\begin{column}{0.54\textwidth}
\begin{itemize}\makebiggaps
\item Earlier cohort ($Z{=}0$): \textbf{0\%} treated, \textbf{20\%} die.
\item Later cohort ($Z{=}1$): \textbf{50\%} treated, \textbf{15\%} die.
\item \emph{If} nothing else moved the death rate ($\delta{=}0$), the whole $5$pp
      drop is the treatment ---
\item concentrated in the $50\%$ who got it: $\;\ATT = -10$pp.
\end{itemize}
\medskip
The \emph{only} lever is $\delta$: how much baseline mortality would have shifted
anyway, absent the treatment change.
\end{column}
\begin{column}{0.44\textwidth}
\includegraphics[width=\linewidth]{SCQEconcept.png}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{SCQE --- the identity}
\small
Cohort $Z\in\{0,1\}$, treatment $D$, $\pi_1 = P(D{=}1\mid Z{=}1)$.
The one untestable quantity is the \textbf{baseline trend}
\[ \delta \;\equiv\; \E[Y(0)\mid Z{=}1] - \E[Y(0)\mid Z{=}0]. \]
For \emph{any} assumed $\delta$, the data pin a unique effect (no-use earlier cohort):
\[ \boxed{\;\ATT(\delta) \;=\; \frac{\Delta\bar Y - \delta}{\pi_1}\;}
   \qquad{\footnotesize\text{(general form: Wulf \& Hazlett et al.\ 2025, Eq.\ 1)}} \]
\begin{itemize}
\item Ignorability (``no confounding'') is \emph{replaced} by a nameable, boundable $\delta$.
\item Sweep $\delta$ over a defensible range $\Rightarrow$ the ATT band ---
      \emph{the same move as Case 2}.
\end{itemize}
\end{frame}

\begin{frame}{You already know two special cases of this}
\small
The ``SCQE plot'' is $\ATT(\delta)$ --- a whole line. Two methods you just learned
are single \emph{rows} of it:
\begin{itemize}\makebiggaps
\item \textbf{Instrumental variables} $=$ SCQE at $\delta = 0$ (time as a ``broken
      instrument''; the $\delta$-adjusted Wald estimand). IV \emph{assumes} the
      baseline trend is exactly zero.
\item \textbf{Difference-in-differences} $=$ SCQE at $\delta = $ the control group's
      observed change. Parallel trends \emph{is} a point assumption on $\delta$.
\end{itemize}
\vfill
\onslide<2->{\textbf{Both pick one $\delta$ and hide it.} SCQE shows the whole line
--- then asks which $\delta$ you can defend.}
\end{frame}

\begin{frame}{Four ways at $\YY$}
\begin{itemize}\makebiggaps
\item \textbf{Know it} --- prognosis near-deterministic; one treated arm nearly as
      informative as an RCT.
\item \textbf{Bound it} --- partial identification: a range you believe.
\item \textbf{Model its trend} --- SCQE: a nameable, boundable baseline trend
      replaces untestable ``no confounding.''
\item \textbf{Stress-test it} --- sensitivity analysis: how wrong would you have to be?
\end{itemize}
\end{frame}

\begin{frame}{You already believe in this}
\small
% REUSE: ../../slides/custom/motivating_cases.tex — smoking & lung cancer bullet
\begin{itemize}\makebiggaps
\item Smoking causes lung cancer --- among the most trusted claims in science.
\item No RCT. Ever. We believe it because someone formalized \emph{what it would
      take} to be wrong --- and the confounding required was absurd.
\end{itemize}
\vfill
\onslide<2->{\textbf{That is sensitivity analysis. That is safe inference.} You just
never called it that.}
\end{frame}

\begin{frame}{Graceful degradation --- a dimmer, not a switch}
\centering
\begin{tikzpicture}[scale=0.95]
  \draw[->] (0,0) -- (6.6,0) node[below right] {\footnotesize how well we pin $\YY$};
  \draw[->] (0,0) -- (0,3.5) node[above] {\footnotesize strength of claim};
  % RCT fundamentalism: flat silence, then a jump to full belief at "RCT"
  \draw[very thick, red!70!black] (0,0.15) -- (4,0.15);
  \draw[very thick, red!70!black, dashed] (4,0.15) -- (4,3);
  \draw[very thick, red!70!black] (4,3) -- (6.2,3);
  \node[red!70!black] at (2,0.5) {\footnotesize silence};
  \node[red!70!black, align=center] at (5.2,3.35) {\footnotesize ``RCT!''};
  % safe inference: smooth ramp
  \draw[very thick, someblue] (0,0.15) .. controls (3.2,1.1) and (4.5,2.2) .. (6.2,3.05);
  \node[someblue] at (2.7,2.1) {\footnotesize safe inference};
\end{tikzpicture}
\vfill
{\footnotesize Fundamentalism: silence until a trial, then full belief --- a
\emph{switch}. Safe inference: the claim strengthens \emph{smoothly} as $\YY$ gets
better pinned down --- a \emph{dimmer}.}
\end{frame}

\begin{frame}{The deep point}
\centering\large
An RCT's value is \emph{proportional to our ignorance of} $\YY$.\\[.6em]
\onslide<2->{\normalsize Where $\YY$ is knowable by other means, randomization is
redundant --- demanding the trial anyway is ritual, not rigor.}
\end{frame}

\begin{frame}{But: \emph{two} conditions --- this is the sharp part}
\begin{enumerate}\makebiggaps
\item<1-> \textbf{Recoverable $\YY$} --- known or defensibly boundable (narrow $\delta$).
\item<2-> \textbf{Signal clears the band} --- the effect is large relative to the
      $\delta$-band width $+$ selection.
\end{enumerate}
\vfill
\onslide<3->{\footnotesize When a knowable-but-noisy $\YY$ meets a \emph{small}
effect, the band straddles zero --- and \emph{that} is exactly when the RCT is
still doing real work. Match the instrument to where $\YY$ sits.}
\end{frame}

\section{The ladder}      % ARC 6 (~14 min) — same engine, escalating, both verdicts

% SOURCE: Hazlett, Maokola & Wulf 2020, Statistics in Medicine. Figure IPT_5 is Chad's own.
\begin{frame}{TB / isoniazid --- \textsc{restraint} \hfill\small(commission)}
\small
Isoniazid preventive therapy (IPT), 26{,}715 HIV patients, Tanzania.
\begin{columns}[T]
\begin{column}{0.44\textwidth}
\begin{itemize}\makebiggaps
\item After IPT arrived: \textbf{16\%} of untreated developed TB vs \textbf{0.5\%}
      of treated.
\item Naive gap: \textbf{15.5pp} lower risk ($p\ll0.001$); covariate adjustment
      barely moves it.
\end{itemize}
\end{column}
\begin{column}{0.54\textwidth}
\includegraphics[width=\linewidth]{IPT_5.jpg}
\end{column}
\end{columns}
\vfill
\onslide<2->{\footnotesize But IPT went disproportionately to \emph{low-risk} patients ---
so the treated's own untreated risk sat far below the 16\% seen among the untreated
(figure: imputed $\sim$4\%, not 16\%).}
\end{frame}

\begin{frame}{TB --- what SCQE reveals}
\small
\begin{itemize}\makebiggaps
\item Assume the baseline trend $\approx 0$ (TB incidence stable, absent treatment):
      the effect is \textbf{small and \emph{not} statistically significant}.
\item To claim IPT \emph{helped}, you must argue non-treatment TB incidence would
      have \textbf{risen $\ge 0.7$pp per year} --- plausible, but far from certain.
\end{itemize}
\vfill
\onslide<2->{\textbf{The 15.5pp ``effect'' was the regression tell} --- a tight interval
that came out the same no matter the assumption. SCQE reins it back to an honest
``we can't be sure it helped.''}
\end{frame}

% ==== GBM FLAGSHIP: DCVax-L. Real analysis from dcvax_memo_v2 (published rates). ====
\begin{frame}{Glioblastoma --- the trial that fell apart \hfill\small(the flagship)}
\small
\textbf{DCVax-L}, a dendritic-cell vaccine (Liau et al., \emph{JAMA Oncology} 2023).
GBM kills in $\sim$13--14 months on standard care. \emph{This is the cold open.}
\begin{itemize}\makebiggaps
\item Started randomized --- then broke, three ways:
      \begin{itemize}\small
        \item primary endpoint (PFS) abandoned (pseudoprogression);
        \item $\sim$90\% \emph{crossover} gutted the placebo arm;
        \item a 640-patient \textbf{external control arm} bolted on post hoc.
      \end{itemize}
\item Critics: the outside controls may be sicker $\Rightarrow$ the ``benefit''
      could be \emph{selection}. So the evidence gets dismissed.
\end{itemize}
\vfill
\onslide<2->{\textbf{Don't dismiss it --- and don't rubber-stamp it. Ask what you'd
have to believe.}}
\end{frame}

\begin{frame}{What would you have to believe? \hfill\small(one-cohort SCQE)}
\scriptsize
$\sim$90\% got the vaccine. Split the sample: \textbf{ever-treated} ($N{\approx}296$)
vs a tiny \textbf{never-treated} group ($N{\approx}35$); published overall 60-mo
survival is $13.0\%$. Now sweep the one thing we can't see --- the sample's
\emph{counterfactual} 60-mo survival $\E[\YY]$, had it gone untreated --- and read the effect.
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{lc}
\hline
Ever-treated 60-mo survival (implied) & $\approx 13.8\%$\\
Never-treated 60-mo survival (assumed; $N{=}35$, barely matters) & $\approx 6\%$\\
External controls (ECA), for reference only & $5.7\%$\\
\hline
\textbf{Significant positive} effect while $\E[\YY]\le$ & $\mathbf{9.4\%}$\\
Point estimate still \textbf{positive} while $\E[\YY]\le$ & $13\%$\\
\textbf{Not} significantly harmful unless $\E[\YY]\ge$ & $17\%$\\
\hline
\end{tabular}
\end{center}
{\tiny $\E[\YY]$ = assumed counterfactual (untreated) 60-mo survival of the trial sample.
The within-trial split is not published; the result is \emph{insensitive} to the
never-treated rate ($N{=}35$). Mock-up on published totals, Liau et al.\ 2023 --- the
real split needs individual patient data.}
\end{frame}

\begin{frame}{The honest verdict}
\small
\begin{itemize}\makebiggaps
\item \textbf{Significant benefit} holds as long as you believe the sample's untreated
      60-mo survival would not have topped $\sim$\textbf{9.4\%} --- comfortably above
      the $5.7\%$ the external controls showed (though, honestly, not by a huge margin).
\item To call the effect \textbf{zero}, you must believe it would have reached
      $\sim$\textbf{13\%} untreated --- essentially what the \emph{treated} achieved.
      That is near-total selection bias.
\item You can be confident it was \textbf{not significantly harmful} unless that
      counterfactual tops $17\%$.
\end{itemize}
\vfill
\onslide<2->{\textbf{That is not ``it worked.''} It is exactly \emph{how much you must
believe} to say it didn't.}\\[.3em]
\onslide<3->{\emph{Safe inference: refusing to dismiss, and refusing to overclaim ---
on the same broken trial.}}
\end{frame}

% SOURCE: Wulf, Hazlett, et al. 2025, Observational Studies 11(3):301-330,
% "Safe inference outside of randomized trials." Figures vendored in figures/.
\begin{frame}{Three therapies, one method --- \emph{before} the trials}
\small
Two hospital systems, early pandemic. Usage of each therapy shifted sharply between
cohorts. SCQE asks only: \emph{how much could COVID mortality have drifted between
cohorts, had that shift not happened?} ($=\delta$)
\begin{itemize}\makebiggaps
\item \textbf{Remdesivir}, \textbf{dexamethasone}, \textbf{hydroxychloroquine}.
\item No RCT results yet. Physicians deciding \emph{now}.
\end{itemize}
\vfill
\onslide<2->{\centering\emph{Same $\delta$-band move. Three different verdicts.}}
\end{frame}

\centering

\end{frame}

\begin{frame}{Dexamethasone --- plausibly beneficial}

\includegraphics[height=0.72\textheight]{scqe_plot_dex_range.png}\\

\begin{frame}{Hydroxychloroquine --- the \textsc{restraint} verdict}

\begin{frame}{The payoff}
\begin{itemize}\makebiggaps
\item Every verdict later \textbf{matched the RCTs} --- rem/dex help; HCQ doesn't.
\item SCQE gave \emph{safe, partial} guidance months early, without ever asserting
      ``no confounding.''
\end{itemize}
\vfill
\onslide<2->{\textbf{\textsc{scale}:} when waiting for the trial is \emph{itself} the
costly choice, silence isn't the safe option --- \emph{calibration} is.}
\end{frame}

\section{The symmetry}    % ARC 7 (~4 min) — the disarming slide

\begin{frame}{Same engine, opposite verdicts}
\centering
\begin{tabular}{ll}
\textsc{restraint} --- rein in an over-claim & TB $\cdot$ HCQ $\cdot$ pancreatic\\[.5em]
\textsc{rescue} --- recover a defensible one & GBM $\cdot$ melanoma $\cdot$ rem/dex\\
\end{tabular}
\vfill
\onslide<2->{\textbf{One $\delta$-band move. The verdict follows the evidence ---
not the analyst's temperament.}}\\[.3em]
\onslide<3->{\emph{This is calibration --- not skepticism, and not boldness.}}
\end{frame}

\begin{frame}{And it's not just medicine}
\small
\begin{itemize}\makebiggaps
\item \textbf{Rescue:} naloxone / harm-reduction laws --- $\YY$ (overdose-mortality
      trend) boundable, effect can clear the band, RCT unthinkable.
\item \textbf{Restraint:} minimum wage, right-to-carry --- $\YY$ boundable, but the
      effect is small vs.\ baseline volatility. Honest humility, not a bold claim.
\item \textbf{Already done:} pet ownership \& wellbeing (SCQE on panel data) ---
      the discipline in action.
\end{itemize}
\end{frame}

\section{The call}        % ARC 9 (~5 min)

\begin{frame}{Norms to pin above a desk}
\small
\begin{enumerate}\makebiggaps
\item State the untestable assumption \emph{before} the estimate, not in the appendix.
\item Report how wrong you'd have to be, to be wrong.
\item Never let a tight interval stand in for a strong assumption.
\item A range you believe beats a point you don't.
\item Do not discard defensible evidence for want of a trial.
\item Reward the honest ``we can't tell yet'' over the confident wrong answer.
\end{enumerate}
\end{frame}

\begin{frame}{What ``rigor'' should mean}
\centering\large
Not the \emph{tightest} interval.\\[.4em]
The \emph{truest} one.
\end{frame}

\begin{frame}[plain]
\centering\large
% Flips the SICSS title "Causality you can believe in?"
Causality you can believe in isn't the causality that sounds most certain ---\\[.5em]
it's the causality still standing after you've told the whole truth\\
about what you assumed.
\end{frame}
